架构设计

概述

Rux 是一个专为 RISC-V 64位架构设计的 Linux 兼容内核，使用 Rust 语言编写。

设计目标

  • POSIX 兼容 - 完全遵循 POSIX 标准
  • Linux ABI 兼容 - 二进制级别兼容 Linux
  • 安全性 - 利用 Rust 的内存安全特性
  • 简洁性 - 清晰的模块化设计

    RAM (rwx) : ORIGIN = 0x80200000, LENGTH = 128M
}

SECTIONS {
    . = 0x80200000;

    .text : { *(.text) }
    .rodata : { *(.rodata) }
    .data : { *(.data) }
    .bss : { *(.bss) }

    . = ALIGN(4096);
    .pagetables : { *(.pagetables) }

    /DISCARD/ : { *(.comment) }
}

陷阱处理流程

// trap.rs
#[no_mangle]
pub extern "C" fn trap_handler() {
    let scause = r_scause();
    let stval = r_stval();
    let sepc = r_sepc();

    match scause {
        // 系统调用
        8 => handle_syscall(),
        // 指令页错误
        12 => handle_page_fault(sepc, stval),
        // 读取页错误
        13 => handle_page_fault(sepc, stval),
        // 写入页错误
        15 => handle_page_fault(sepc, stval),
        // 定时器中断
        0x8000000000000005 => handle_timer(),
        // 外部中断
        0x8000000000000009 => handle_external(),
        _ => panic!("Unknown trap: {}", scause),
    }
}
