启动流程

概述

Rux 内核的启动流程遵循 RISC-V 规范，从 M-mode 的 OpenSBI 固件开始，最终进入 S-mode 的内核。

Boot Flow Overview: ┌─────────────────────────────────────────────────────────┐ │ QEMU/ Hardware│ └─────────────────────────────────────────────────────────┘ │ ▼ M-mode ┌─────────────────────────────────────────────────────────┐ │ OpenSBI Firmware│ │ - Initialize hardware│ │ - Setup PMP│ │ - Jump to kernel│ └─────────────────────────────────────────────────────────┘ │ ▼ S-mode ┌─────────────────────────────────────────────────────────┐ │ boot.S (Assembly)│ │ - Setup stack│ │ - Clear BSS│ │ - Call _start│ └─────────────────────────────────────────────────────────┘ │ ▼ ┌─────────────────────────────────────────────────────────┐ │ main.rs - kernel_init() │ │ - Parse FDT│ │ - Initialize subsystems│ │ - Start init process│ └─────────────────────────────────────────────────────────┘

OpenSBI

OpenSBI 简介

OpenSBI 是 RISC-V 的开源监督二进制接口实现，运行在 M-mode，为 S-mode 内核提供：

  • 硬件初始化
  • SBI (Supervisor Binary Interface) 调用
  • 异常代理

QEMU 启动参数

qemu-system-riscv64 \
    -machine virt \
    -cpu rv64 \
    -m 2G \
    -bios default\          # 使用 OpenSBI
    -kernel rux.elf \       # 内核镜像
    -append "console=ttyS0" \
    -nographic

boot.S 启动汇编

入口点

// kernel/src/arch/riscv64/boot.S

.section .text.init
.global _start

_start:
    // 1. 关闭中断
    csrw sie, zero
    csrw sip, zero

    // 2. 设置栈指针
    la sp, _stack_top

    // 3. 清除 BSS 段
    la t0, _bss_start
    la t1, _bss_end
clear_bss:
    bgeu t0, t1, 2f
    sd zero, 0(t0)
    addi t0, t0, 8
    j clear_bss
2:

    // 4. 跳转到 Rust 代码
    call kernel_init

    // 不应该到达这里
3:  wfi
    j 3b

栈设置

// 栈大小定义
#define STACK_SIZE (64 * 1024)  // 64KB per CPU

.section .bss
.align 16
_stack_bottom:
    .space STACK_SIZE
_stack_top:

    #address-cells = <2>;
    #size-cells = <2>;

    // 内存信息

内存区域解析

// cmdline.rs
pub unsafe fn parse_memory_regions(dtb: *const u8) -> Vec {
    let mut regions = Vec::new();

    // 遍历 memory 节点
    for_each_node(dtb, |node| {
        if node.name.starts_with("memory@") {
            let reg = parse_reg_property(node);
            regions.push(MemoryRegion {
                base: reg.address,
                size: reg.size,
            });
        }
    });

    regions
}

内核初始化

初始化序列

// main.rs
#[no_mangle]
pub extern "C" fn kernel_init(hartid: usize, dtb_ptr: *const u8) {
    // 1. UART 初始化 (最早，用于调试输出)
    arch::uart::init();
    println!("Rux Kernel starting...");

    // 2. 解析设备树
    cmdline::parse_fdt(dtb_ptr);

    // 3. 初始化 memblock
    mm::memblock_init();
    let memory_regions = unsafe { cmdline::parse_memory_regions(dtb_ptr) };
    for region in &memory_regions {
        mm::memblock_add(region.base, region.size).ok();
    }

    // 4. 保留已使用区域
    mm::memblock_reserve(0x80000000, 0xA00000).ok();  // OpenSBI + kernel

    // 5. 内存管理初始化
    arch::mm::init();
    let frame_start = mm::memblock_get_available_region()
        .map(|r| r.base).unwrap_or(0x88000000);
    mm::page::init_frame_allocator(frame_start / PAGE_SIZE);

    // 6. 堆初始化
    heap::init();

    // 7. 中断和陷阱初始化
    arch::trap::init();

    // 8. 调度器初始化
    scheduler::init();

    // 9. 文件系统初始化
    fs::init();

    // 10. 创建 init 进程
    init::create_init_process();

    // 11. 启动调度器
    scheduler::run();
}
