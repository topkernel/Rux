任务结构

Task 结构体

pub struct Task {
    // 进程标识
    pub pid: usize,
    pub tid: usize,           // 线程 ID
    pub ppid: usize,          // 父进程 ID
    pub state: TaskState,
    pub exit_code: i32,

    // 内存管理
    pub mm: Arc,    // 地址空间

    // 调度信息
    pub priority: usize,
    pub time_slice: usize,
    pub need_resched: bool,

    // 上下文
    pub context: TaskContext,

    // 文件系统
    pub files: Arc,  // 文件描述符表
    pub fs_info: Arc, // 文件系统信息

    // 信号
    pub signals: SignalSet,
    pub sigaction: [SigAction; 64],
}

进程状态

    Blocked,        // 阻塞
    Zombie,         // 僵尸
    Stopped,        // 停止
}

上下文切换

TaskContext 结构

pub struct TaskContext {
    pub ra: u64,          // 返回地址
    pub sp: u64,          // 栈指针

}

RISC-V 汇编实现

// context_switch.S
__switch:
    // Save prev context
    sd ra, 0(a0)
    sd sp, 8(a0)
    sd s0, 16(a0)
    // ... save s1-s11

    // Restore next context
    ld ra, 0(a1)
    ld sp, 8(a1)
    ld s0, 16(a1)
    // ... restore s1-s11

    ret

调度器

调度策略

目前实现了简单的优先级调度：

pub fn schedule() {
    let next = select_next_task();
    if next.pid != current.pid {
        context_switch(current, next);
    }
}

fn select_next_task() -> Arc {
    // 从就绪队列中选择最高优先级任务
    READY_QUEUE.lock()
        .iter()
        .filter(|t| t.state == TaskState::Ready)
        .max_by_key(|t| t.priority)
        .cloned()
        .unwrap_or(IDLE_TASK.clone())
}

调度时机

  • 时间片用完
  • 进程主动让出 CPU (sched_yield)
  • 进程阻塞（等待 I/O、信号等）
  • 进程退出

ELF 程序加载

execve 系统调用

pub fn sys_execve(path: &str, argv: &[&str], envp: &[&str]) -> isize {
    // 1. 读取 ELF 文件
    let elf = load_elf(path)?;

    // 2. 创建新的地址空间
    let new_mm = create_user_address_space();

    // 3. 加载段到内存
    for phdr in elf.program_headers {
        if phdr.p_type == PT_LOAD {
            load_segment(&new_mm, &phdr)?;
        }
    }

    // 4. 设置栈
    setup_user_stack(&new_mm, argv, envp)?;

    // 5. 切换到新地址空间
    current.mm = new_mm;

    // 6. 跳转到入口点
    set_pc(elf.entry);
}
